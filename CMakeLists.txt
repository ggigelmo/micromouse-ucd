cmake_minimum_required(VERSION 3.16)
project(micromouse_ucd LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(solver src/MazeMap.cpp src/FloodFill.cpp src/MouseAgent.cpp)
target_include_directories(solver PUBLIC src)
add_executable(micromouse Main.cpp API.cpp)
target_link_libraries(micromouse PRIVATE solver)
add_executable(solver_tests tests/solver_tests.cpp)
target_link_libraries(solver_tests PRIVATE solver)
if(MSVC)
  target_compile_options(solver_tests PRIVATE /UNDEBUG)
else()
  target_compile_options(solver_tests PRIVATE -UNDEBUG)
endif()
enable_testing()
add_test(NAME solver_and_motion_math COMMAND solver_tests)
